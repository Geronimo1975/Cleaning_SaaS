# Modulul de comenzi (Customer + Order) — plan de implementare

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Un app `apps/billing` cu `Customer`, `Order`, `DocumentSequence`, creat automat la activarea unei cereri din marketplace, vizibil în dashboard la `/dashboard/comenzi/`, cu backfill pentru cererile deja active.

**Architecture:** Un singur app nou, `apps/billing`, cu modele + `numbering.py` (numere sub lock) + `services.py` (creare din cerere, tranziții de stare) + `backfill.py` (folosit de migrația de date). Seam-ul e un apel local în `activate_recurring_request`. Views server-rendered în dashboard, gate prin `CapabilityRequiredMixin` + `OwnedByAccountMixin(account_field="provider")`.

**Tech Stack:** Django 5.2.17, Python 3.12, pytest-django (`--reuse-db`), Postgres (SQLite local).

**Spec:** `docs/superpowers/specs/2026-09-12-comenzi-design.md`

```python
"""
Stratul comercial, partea de comenzi.

`Customer` e o RELAȚIE între două conturi (prestator ↔ beneficiar), nu o a doua
tabelă de părți: USt-IdNr., adresa, forma juridică rămân pe `accounts.Account`.
`Order` (Auftrag) e containerul de care se vor lega ofertele și facturile.
`DocumentSequence` ține numerotarea per prestator, per tip, per an.
"""

from django.core.exceptions import ValidationError
from django.db import models

from .enums import ContractKind, DocumentType, OrderStatus


class Customer(models.Model):
    provider = models.ForeignKey(
        "accounts.Account", on_delete=models.PROTECT, related_name="customers",
        verbose_name="Prestator",
    )
    account = models.ForeignKey(
        "accounts.Account", on_delete=models.PROTECT, related_name="customer_of",
        verbose_name="Beneficiar",
    )
    customer_number = models.CharField("Număr client", max_length=32, blank=True)
    contact_person = models.CharField("Persoană de contact", max_length=160, blank=True)
    is_active = models.BooleanField("Activ", default=True)
    created_at = models.DateTimeField(auto_now_add=True)

    class Meta:
        ordering = ("account__display_name",)
        constraints = [
            models.UniqueConstraint(fields=("provider", "account"), name="uniq_customer"),
            models.CheckConstraint(
                condition=~models.Q(provider=models.F("account")), name="customer_not_self"
            ),
        ]
        verbose_name = "Client"
        verbose_name_plural = "Clienți"

    def __str__(self):
        return f"{self.account} @ {self.provider}"


class DocumentSequence(models.Model):
    """Numerotare per prestator, per tip de document, per an. Se alocă sub lock
    în `numbering.allocate()`, niciodată din `pk`."""

    provider = models.ForeignKey(
        "accounts.Account", on_delete=models.CASCADE, related_name="document_sequences"
    )
    document_type = models.CharField(max_length=16, choices=DocumentType.choices)
    year = models.PositiveSmallIntegerField()
    prefix = models.CharField("Prefix", max_length=8, default="A")
    next_number = models.PositiveIntegerField("Următorul număr", default=1)

    class Meta:
        constraints = [
            models.UniqueConstraint(
                fields=("provider", "document_type", "year"), name="uniq_sequence"
            ),
        ]
        verbose_name = "Serie de numerotare"
        verbose_name_plural = "Serii de numerotare"

    def __str__(self):
        return f"{self.prefix}-{self.year} → {self.next_number}"


class Order(models.Model):
    """Auftrag. Obiectivul (`planning.Site`) există deja — nu se dublează aici."""

    provider = models.ForeignKey(
        "accounts.Account", on_delete=models.PROTECT, related_name="orders",
        verbose_name="Prestator",
    )
    customer = models.ForeignKey(
        Customer, on_delete=models.PROTECT, related_name="orders", verbose_name="Client"
    )
    site = models.ForeignKey(
        "planning.Site", on_delete=models.PROTECT, related_name="orders",
        verbose_name="Obiectiv",
    )
    category = models.ForeignKey(
        "taxonomy.ServiceCategory", on_delete=models.PROTECT, related_name="orders",
        verbose_name="Tip de serviciu",
    )
    order_number = models.CharField("Număr comandă", max_length=32)
    title = models.CharField("Titlu", max_length=200)
    start_date = models.DateField("Început")
    end_date = models.DateField("Sfârșit", null=True, blank=True)
    contract_kind = models.CharField(
        "Natura contractului", max_length=16, choices=ContractKind.choices,
        default=ContractKind.DIENSTVERTRAG,
    )
    # Abnahme (§641 BGB). Are sens doar la Werkvertrag.
    accepted_at = models.DateTimeField("Recepție", null=True, blank=True)
    status = models.CharField(
        "Stare", max_length=12, choices=OrderStatus.choices,
        default=OrderStatus.ACTIVE, db_index=True,
    )
    source_offer = models.ForeignKey(
        "marketplace.Offer", null=True, blank=True, on_delete=models.SET_NULL,
        related_name="orders", verbose_name="Oferta câștigătoare",
    )
    created_at = models.DateTimeField(auto_now_add=True)

    class Meta:
        ordering = ("-start_date", "-pk")
        constraints = [
            models.UniqueConstraint(
                fields=("provider", "order_number"), name="uniq_order_number_per_provider"
            ),
        ]
        verbose_name = "Comandă"
        verbose_name_plural = "Comenzi"

    def __str__(self):
        return f"{self.order_number} — {self.title}"

    def clean(self):
        errors = {}
        if self.site_id and self.site.account_id != self.provider_id:
            errors["site"] = "Obiectivul aparține altui cont."
        if self.customer_id and self.customer.provider_id != self.provider_id:
            errors["customer"] = "Clientul aparține altui prestator."
        if self.category_id and not self.category.parent_id:
            errors["category"] = "Doar categoriile-frunză sunt selectabile."
        if self.start_date and self.end_date and self.end_date < self.start_date:
            errors["end_date"] = "Sfârșitul nu poate fi înaintea începutului."
        if errors:
            raise ValidationError(errors)
```

```python
        numbers = sorted(pool.map(lambda _: worker(), range(20)))
```

```python
"""
Numere de document fără goluri. Se alocă DOAR în tranzacția care creează
documentul, sub `select_for_update` — niciodată din `pk`, niciodată reutilizate.
Lock-ul pe rând ține pe Postgres; SQLite (doar local) serializează oricum.
"""

from django.db import transaction

from .models import DocumentSequence


@transaction.atomic
def allocate(provider, document_type: str, year: int) -> str:
    DocumentSequence.objects.get_or_create(
        provider=provider, document_type=document_type, year=year
    )
    seq = DocumentSequence.objects.select_for_update().get(
        provider=provider, document_type=document_type, year=year
    )
    number = f"{seq.prefix}-{year}-{seq.next_number:04d}"
    seq.next_number += 1
    seq.save(update_fields=["next_number"])
    return number
```

```python
"""
Logica de comenzi. Comanda se naște din cererea activată în marketplace —
un singur drum, folosit și de backfill — și își schimbă starea doar pe aici.
"""

from django.db import transaction
from django.utils import timezone

from .enums import ContractKind, DocumentType, OrderStatus
from .models import Customer, Order
from .numbering import allocate


class BillingError(ValueError):
    pass


ALLOWED_TRANSITIONS: dict[str, set[str]] = {
    OrderStatus.ACTIVE: {OrderStatus.SUSPENDED, OrderStatus.COMPLETED, OrderStatus.CANCELLED},
    OrderStatus.SUSPENDED: {OrderStatus.ACTIVE, OrderStatus.CANCELLED},
    OrderStatus.COMPLETED: set(),
    OrderStatus.CANCELLED: set(),
}


@transaction.atomic
def create_order_from_request(request, site, *, actor) -> Order:
    """
    Customer + Order pentru o cerere ACTIVĂ. Idempotent prin `source_offer`:
    a doua chemare întoarce comanda existentă, nu creează alta.
    """
    from apps.marketplace.models import RequestStatus

    offer = request.awarded_offer
    if request.status != RequestStatus.ACTIVE or offer is None:
        raise BillingError("Comanda se creează doar dintr-o cerere activată.")

    existing = Order.objects.filter(source_offer=offer).first()
    if existing is not None:
        return existing

    customer, _ = Customer.objects.get_or_create(provider=offer.provider, account=request.client)
    order = Order(
        provider=offer.provider,
        customer=customer,
        site=site,
        category=request.category,
        order_number=allocate(offer.provider, DocumentType.ORDER, offer.earliest_start.year),
        title=request.title[:200],
        start_date=offer.earliest_start,
        source_offer=offer,
    )
    order.full_clean()
    order.save()
    return order


def transition_order(order: Order, new_status: str, *, actor) -> Order:
    if actor.pk != order.provider_id:
        raise BillingError("Doar prestatorul își schimbă comenzile.")
    if new_status not in ALLOWED_TRANSITIONS[order.status]:
        raise BillingError(
            f"Din „{order.get_status_display()}” nu se poate trece la „{new_status}”."
        )
    fields = ["status"]
    order.status = new_status
    if (
        new_status == OrderStatus.COMPLETED
        and order.contract_kind == ContractKind.WERKVERTRAG
        and order.accepted_at is None
    ):
        order.accepted_at = timezone.now()   # Abnahme, §641 BGB
        fields.append("accepted_at")
    order.save(update_fields=fields)
    return order
```

```python
"""
Backfill: cererile ACTIVE de dinaintea modulului de comenzi primesc Customer +
Order, cu aceeași cheie de obiectiv pe care o folosește activarea
(`Site(account=offer.provider, name=request.title[:160])`).

Lucrează pe modelele istorice primite de la migrație, deci nu importă
`services.py`. Migrațiile rulează serial — nu e nevoie de `select_for_update`.
"""


def backfill_orders(apps) -> int:
    ServiceRequest = apps.get_model("marketplace", "ServiceRequest")
    Site = apps.get_model("planning", "Site")
    Customer = apps.get_model("billing", "Customer")
    Order = apps.get_model("billing", "Order")
    DocumentSequence = apps.get_model("billing", "DocumentSequence")

    created = 0
    requests = (
        ServiceRequest.objects.filter(status="ACTIVE", awarded_offer__isnull=False)
        .select_related("awarded_offer", "client", "category")
        .order_by("pk")
    )
    for request in requests:
        offer = request.awarded_offer
        if Order.objects.filter(source_offer=offer).exists():
            continue
        site = Site.objects.filter(account_id=offer.provider_id, name=request.title[:160]).first()
        if site is None:
            continue
        customer, _ = Customer.objects.get_or_create(
            provider_id=offer.provider_id, account_id=request.client_id
        )
        year = offer.earliest_start.year
        seq, _ = DocumentSequence.objects.get_or_create(
            provider_id=offer.provider_id, document_type="ORDER", year=year
        )
        number = f"{seq.prefix}-{year}-{seq.next_number:04d}"
        seq.next_number += 1
        seq.save(update_fields=["next_number"])
        Order.objects.create(
            provider_id=offer.provider_id,
            customer=customer,
            site=site,
            category_id=request.category_id,
            order_number=number,
            title=request.title[:200],
            start_date=offer.earliest_start,
            source_offer=offer,
        )
        created += 1
    return created
```

```python
"""Comenzi și clienți, văzute de prestator. Scoping strict pe `provider`."""

from django.contrib import messages
from django.shortcuts import get_object_or_404, redirect
from django.views.decorators.http import require_POST
from django.views.generic import DetailView, ListView

from apps.accounts.enums import Capability
from apps.accounts.permissions import (
    AccountRequiredMixin,
    CapabilityRequiredMixin,
    OwnedByAccountMixin,
    require_manage,
)

from .models import Customer, Order
from .services import ALLOWED_TRANSITIONS, BillingError, transition_order

PROVIDER_SIDE = (Capability.PROVIDER, Capability.WORKER)


class CustomerListView(CapabilityRequiredMixin, OwnedByAccountMixin, ListView):
    required_capability = PROVIDER_SIDE
    account_field = "provider"
    model = Customer
    template_name = "dashboard/customer_list.html"
    context_object_name = "customers"

    def get_queryset(self):
        return super().get_queryset().select_related("account__locality")


class OrderListView(CapabilityRequiredMixin, OwnedByAccountMixin, ListView):
    required_capability = PROVIDER_SIDE
    account_field = "provider"
    model = Order
    template_name = "dashboard/order_list.html"
    context_object_name = "orders"
    paginate_by = 50

    def get_queryset(self):
        return super().get_queryset().select_related("customer__account", "site", "category")


class OrderDetailView(CapabilityRequiredMixin, OwnedByAccountMixin, DetailView):
    required_capability = PROVIDER_SIDE
    account_field = "provider"
    model = Order
    template_name = "dashboard/order_detail.html"
    context_object_name = "order"

    def get_queryset(self):
        return super().get_queryset().select_related(
            "customer__account", "site__locality", "category", "source_offer"
        )

    def get_context_data(self, **kwargs):
        ctx = super().get_context_data(**kwargs)
        labels = dict(Order._meta.get_field("status").choices)
        ctx["next_statuses"] = [
            (value, labels[value]) for value in sorted(ALLOWED_TRANSITIONS[self.object.status])
        ]
        return ctx


@require_POST
def order_transition_view(request, pk):
    if request.account is None:
        return redirect("accounts:onboarding")
    if not (set(PROVIDER_SIDE) & request.account.capability_set()):
        from django.core.exceptions import PermissionDenied

        raise PermissionDenied
    order = get_object_or_404(Order, pk=pk, provider=request.account)
    require_manage(request.membership)
    try:
        transition_order(order, request.POST.get("status", ""), actor=request.account)
        messages.success(request, f"Comanda {order.order_number}: {order.get_status_display()}.")
    except BillingError as exc:
        messages.error(request, str(exc))
    return redirect("dashboard:order_detail", pk=order.pk)
```
